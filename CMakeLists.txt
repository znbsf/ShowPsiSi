cmake_minimum_required(VERSION 3.22)
project(ShowPsiSi VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(BUILD_QT_UI "Build Qt Quick prototype" ON)
set(TSDUCK_ROOT "C:/Program Files/TSDuck" CACHE PATH "Installed TSDuck developer SDK")
find_path(TSDUCK_INCLUDE tsduck.h HINTS "${TSDUCK_ROOT}/include" PATH_SUFFIXES tsduck)
find_library(TSDUCK_LIB NAMES tsduck HINTS "${TSDUCK_ROOT}/lib/Release-Win64")
if(NOT TSDUCK_INCLUDE OR NOT TSDUCK_LIB)
  message(FATAL_ERROR "TSDuck developer library missing. Set TSDUCK_ROOT.")
endif()
add_executable(showpsisi-core core/main.cpp)
target_include_directories(showpsisi-core PRIVATE ${TSDUCK_INCLUDE} third_party)
target_link_libraries(showpsisi-core PRIVATE ${TSDUCK_LIB})
if(WIN32)
  target_compile_definitions(showpsisi-core PRIVATE _TSDUCKDLL_USE NOMINMAX)
  add_custom_command(TARGET showpsisi-core POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${TSDUCK_ROOT}/bin/tsduck.dll" "$<TARGET_FILE_DIR:showpsisi-core>")
endif()
if(MSVC)
  add_compile_options(/utf-8 /W3)
  target_compile_options(showpsisi-core PRIVATE /utf-8 /W3)
endif()
if(BUILD_QT_UI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
  qt_standard_project_setup(REQUIRES 6.5)
  qt_add_executable(showpsisi-qt ui/qt/main.cpp)
  qt_add_qml_module(showpsisi-qt URI ShowPsiSi VERSION 1.0 QML_FILES ui/qt/Main.qml)
  target_link_libraries(showpsisi-qt PRIVATE Qt6::Quick Qt6::QuickControls2)
  set_target_properties(showpsisi-qt PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
